4. CẤU TRÚC DATABASE (FIRESTORE SCHEMA)
================================================================

Collection: users
─────────────────
users/{uid}
  ├── uid         : String
  ├── name        : String
  ├── email       : String
  ├── phone       : String
  ├── avatarUrl   : String
  ├── fcmToken    : String       ← cập nhật mỗi lần login
  └── createdAt   : Timestamp

Collection: movies
──────────────────
movies/{movieId}
  ├── movieId     : String
  ├── title       : String
  ├── description : String
  ├── genre       : List<String>   e.g. ["Action","Sci-Fi"]
  ├── duration    : int            (phút)
  ├── rating      : float          (e.g. 8.5)
  ├── ageLimit    : String         ("P", "T13", "T16", "T18")
  ├── posterUrl   : String
  ├── trailerUrl  : String
  ├── language    : String
  ├── releaseDate : Timestamp
  ├── isNowShowing: boolean
  └── isComingSoon: boolean

Collection: theaters
────────────────────
theaters/{theaterId}
  ├── theaterId   : String
  ├── name        : String         e.g. "CGV Vincom Ba Trieu"
  ├── address     : String
  ├── city        : String
  ├── latitude    : double
  ├── longitude   : double
  └── imageUrl    : String

Collection: showtimes
─────────────────────
showtimes/{showtimeId}
  ├── showtimeId  : String
  ├── movieId     : String
  ├── theaterId   : String
  ├── roomName    : String         e.g. "Screen 3 - IMAX"
  ├── startTime   : Timestamp
  ├── endTime     : Timestamp
  ├── price       : double
  ├── totalSeats  : int
  ├── availableSeats: int
  └── seatMap     : Map<String, String>
       key = seatId (e.g. "A1", "B5")
       value = status ("available" | "booked" | "selected")

Collection: tickets
───────────────────
tickets/{ticketId}
  ├── ticketId    : String
  ├── userId      : String
  ├── movieId     : String
  ├── showtimeId  : String
  ├── theaterId   : String
  ├── seats       : List<String>   e.g. ["C3","C4"]
  ├── totalPrice  : double
  ├── bookingTime : Timestamp
  ├── status      : String         ("confirmed" | "cancelled" | "used")
  ├── qrCode      : String         (encoded ticket info)
  └── notifSent   : boolean

Firestore Rules (firestore.rules):
───────────────────────────────────
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }
    match /movies/{movieId} {
      allow read: if true;
      allow write: if false; // chỉ admin
    }
    match /theaters/{theaterId} {
      allow read: if true;
    }
    match /showtimes/{showtimeId} {
      allow read: if true;
      allow write: if request.auth != null; // update seatMap khi đặt vé
    }
    match /tickets/{ticketId} {
      allow read: if request.auth != null &&
                  resource.data.userId == request.auth.uid;
      allow create: if request.auth != null;
    }
  }
}

// Dùng Firestore Transaction để đảm bảo atomic
db.runTransaction(transaction -> {
    DocumentReference showtimeRef = db.collection("showtimes").document(showtimeId);
    DocumentSnapshot snapshot = transaction.get(showtimeRef);
    
    // Kiểm tra ghế còn trống
    Map<String, Object> seatMap = (Map<String, Object>) snapshot.get("seatMap");
    for (String seat : selectedSeats) {
        if (!"available".equals(seatMap.get(seat))) {
            throw new FirebaseFirestoreException("Ghế đã được đặt",
                FirebaseFirestoreException.Code.ABORTED);
        }
    }
    
    // Cập nhật trạng thái ghế
    for (String seat : selectedSeats) {
        seatMap.put(seat, "booked");
    }
    transaction.update(showtimeRef, "seatMap", seatMap);
    transaction.update(showtimeRef, "availableSeats",
        snapshot.getLong("availableSeats") - selectedSeats.size());
    
    // Tạo ticket
    String ticketId = db.collection("tickets").document().getId();
    Map<String, Object> ticket = new HashMap<>();
    ticket.put("ticketId", ticketId);
    ticket.put("userId", currentUserId);
    ticket.put("movieId", movieId);
    ticket.put("showtimeId", showtimeId);
    ticket.put("seats", selectedSeats);
    ticket.put("totalPrice", totalPrice);
    ticket.put("bookingTime", FieldValue.serverTimestamp());
    ticket.put("status", "confirmed");
    ticket.put("notifSent", false);
    transaction.set(db.collection("tickets").document(ticketId), ticket);
    
    return ticketId;
}).addOnSuccessListener(ticketId -> {
    // Chuyển sang màn hình vé
    scheduleNotification(ticketId, showtimeTime);
    Intent intent = new Intent(this, TicketDetailActivity.class);
    intent.putExtra("ticketId", ticketId);
    startActivity(intent);
}).addOnFailureListener(e -> {
    Toast.makeText(this, "Đặt vé thất bại: " + e.getMessage(),
        Toast.LENGTH_SHORT).show();

public class FCMService extends FirebaseMessagingService {
    
    @Override
    public void onNewToken(@NonNull String token) {
        super.onNewToken(token);
        // Cập nhật token vào Firestore khi token thay đổi
        String userId = FirebaseAuth.getInstance().getCurrentUser() != null
            ? FirebaseAuth.getInstance().getCurrentUser().getUid() : null;
        if (userId != null) {
            FirebaseFirestore.getInstance()
                .collection("users").document(userId)
                .update("fcmToken", token);
        }
    }

    @Override
    public void onMessageReceived(@NonNull RemoteMessage message) {
        super.onMessageReceived(message);
        String title = message.getNotification().getTitle();
        String body  = message.getNotification().getBody();
        String ticketId = message.getData().get("ticketId");
        NotificationHelper.showNotification(this, title, body, ticketId);
    }
}

public class NotificationHelper {
    public static void showNotification(Context ctx, String title,
                                        String body, String ticketId) {
        // Tạo PendingIntent mở TicketDetailActivity
        Intent intent = new Intent(ctx, TicketDetailActivity.class);
        intent.putExtra("ticketId", ticketId);
        PendingIntent pi = PendingIntent.getActivity(ctx, 0, intent,
            PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);

        NotificationCompat.Builder builder = new NotificationCompat.Builder(
            ctx, "movie_channel")
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(title)
            .setContentText(body)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setAutoCancel(true)
            .setContentIntent(pi)
            .setColor(ContextCompat.getColor(ctx, R.color.primary_red));

        NotificationManagerCompat.from(ctx)
            .notify(ticketId.hashCode(), builder.build());
    }
    
    public static void createChannel(Context ctx) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            NotificationChannel channel = new NotificationChannel(
                "movie_channel", "Movie Notifications",
                NotificationManager.IMPORTANCE_HIGH);
            channel.setDescription("Nhắc nhở lịch chiếu phim");
            ctx.getSystemService(NotificationManager.class)
                .createNotificationChannel(channel);
        }
    }
}

// Gửi thông báo nhắc 30 phút trước suất chiếu
public static void scheduleNotification(String ticketId, long showtimeMillis) {
    long delay = showtimeMillis - System.currentTimeMillis() - 30 * 60 * 1000;
    if (delay <= 0) return;

    Data data = new Data.Builder()
        .putString("ticketId", ticketId)
        .build();

    OneTimeWorkRequest work = new OneTimeWorkRequest.Builder(
        NotificationWorker.class)
        .setInitialDelay(delay, TimeUnit.MILLISECONDS)
        .setInputData(data)
        .addTag(ticketId)
        .build();

    WorkManager.getInstance(context).enqueue(work);
}

// → doWork() → gọi NotificationHelper.showNotification()
